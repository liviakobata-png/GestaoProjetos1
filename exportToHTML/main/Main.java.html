68       private static void menuCadastrarProjeto() {
69           System.out.println("\n--- CADASTRO DE PROJETO ---");
70           List<Usuario> gerentes = service.listarGerentes();
71   
72           if (gerentes.isEmpty()) {
73               System.out.println("❌ Não é possível criar um projeto. Cadastre ao menos um GERENTE primeiro.");
74               return;
75           }
76   
77           System.out.print("Nome do Projeto: "); String nome = scanner.nextLine();
78           System.out.print("Descrição: "); String descricao = scanner.nextLine();
79           System.out.print("Data de Início (DD/MM/AAAA): "); String inicio = scanner.nextLine();
80           System.out.print("Data de Término Prevista (DD/MM/AAAA): "); String termino = scanner.nextLine();
81   
82           System.out.println("Status: 1-Planejado | 2-Em Andamento | 3-Concluído | 4-Cancelado");
83           System.out.print("Escolha: ");
84           int statusOp = Integer.parseInt(scanner.nextLine());
85           StatusProjeto status = StatusProjeto.PLANEJADO;
86           if (statusOp == 2) status = StatusProjeto.EM_ANDAMENTO;
87           else if (statusOp == 3) status = StatusProjeto.CONCLUIDO;
88           else if (statusOp == 4) status = StatusProjeto.CANCELADO;
89   
90           System.out.println("Selecione o Gerente Responsável:");
91           for (int i = 0; i < gerentes.size(); i++) {
92               System.out.println(i + " - " + gerentes.get(i).getNomeCompleto());
93           }
94           System.out.print("Escolha o número do gerente: ");
95           int gerenteIndex = Integer.parseInt(scanner.nextLine());
96           Usuario gerente = gerentes.get(gerenteIndex);
97   
98           Projeto novoProjeto = new Projeto(nome, descricao, inicio, termino, status, gerente);
99           service.cadastrarProjeto(novoProjeto);
100          System.out.println("✅ Projeto cadastrado com sucesso!");
101      }

103      private static void menuCadastrarEquipe() {
104          System.out.println("\n--- CADASTRO DE EQUIPE ---");
105          List<Usuario> todosUsuarios = service.listarUsuarios();
106  
107          if (todosUsuarios.isEmpty()) {
108              System.out.println("❌ Cadastre usuários antes de criar uma equipe.");
109              return;
110          }
111  
112          System.out.print("Nome da Equipe: "); String nome = scanner.nextLine();
113          System.out.print("Descrição da Equipe: "); String descricao = scanner.nextLine();
114  
115          Equipe novaEquipe = new Equipe(nome, descricao);
116  
117          System.out.println("--- Alocação de Membros (Digite -1 para encerrar a seleção) ---");
118          while (true) {
119              for (int i = 0; i < todosUsuarios.size(); i++) {
120                  System.out.println(i + " - " + todosUsuarios.get(i).getNomeCompleto() + " (" + todosUsuarios.get(i).getPerfil() + ")");
121              }
122              System.out.print("Selecione o ID do usuário para adicionar à equipe: ");
123              int id = Integer.parseInt(scanner.nextLine());
124  
125              if (id == -1) break;
126  
127              if (id >= 0 && id < todosUsuarios.size()) {
128                  novaEquipe.adicionarMembro(todosUsuarios.get(id));
129                  System.out.println("➕ " + todosUsuarios.get(id).getNomeCompleto() + " adicionado!");
130              } else {
131                  System.out.println("ID inválido.");
132              }
133              System.out.println();
134          }
135  
136          service.cadastrarEquipe(novaEquipe);
137          System.out.println("✅ Equipe registrada com sucesso!");
138      }
